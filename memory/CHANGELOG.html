aee301c 11:15 BUG-007 已修复
▶
main/session/session.c
+30-17
修改前（立即激活 VAD）
// 收到服务端 "started" 后
// 立即开始 VAD 检测
s_vad_active = true;
// VAD 检测逻辑
if (vad_state == AFE_VAD_SILENCE) {
start_eos_timer();
}
修改后（500ms 消退保护期）
// 收到服务端 "started" 后
s_vad_ready_tick = xTaskGetTickCount();
s_stop_sent = false;
// VAD 检测逻辑
TickType_t elapsed =
xTaskGetTickCount() - s_vad_ready_tick;
bool grace_over =
elapsed > pdMS_TO_TICKS(VAD_GRACE_MS); // 500ms
if (grace_over && vad == AFE_VAD_SILENCE) {
if (!s_stop_sent) {
send_stop(); s_stop_sent = true;
}
}
问题根因：唤醒词尾音 + 呼吸停顿被立即激活的 VAD 判定为句末静音，发送 stop 打断了用户真正的问话。
修复方案：服务端 started 响应后等待 500ms 才开始 VAD 检测，消退期内静音不触发 EOS。
